```python
# hp3458a_cal_dump.py — hozzáfűzni

class GpibSession:
    def __init__(self, resource: str = GPIB_RESOURCE, timeout_ms: int = 5000):
        self._resource   = resource
        self._timeout_ms = timeout_ms
        self._rm         = None
        self._inst       = None

    def __enter__(self) -> 'GpibSession':
        self._rm   = pyvisa.ResourceManager()
        self._inst = self._rm.open_resource(self._resource)
        self._inst.timeout          = self._timeout_ms
        self._inst.read_termination = ''   # csak EOI zár le, ne a termchar
        self._init()
        return self

    def __exit__(self, *_) -> None:
        if self._inst:
            try:
                for cmd in ('BEEP 0', 'TARM AUTO', 'DISP 1'):
                    self._inst.write(cmd)
            except Exception:
                pass
            self._inst.close()
        if self._rm:
            self._rm.close()

    def _init(self) -> None:
        for cmd in ('PRESET NORM', 'END ALWAYS', 'TARM HOLD', 'INBUF ON', 'DISP 2'):
            self._inst.write(cmd)
        for _ in range(10):                        # hibasor kiürítése
            resp = self._inst.query('ERRSTR?').strip()
            if resp.startswith('+0,') or resp == '+0':
                break

    def beep(self, mode: int) -> None:
        self._inst.write(f'BEEP {mode}')

    def mread_word(self, addr: int) -> int:
        self._inst.write(f'MADDR {addr}')
        time.sleep(MREAD_DELAY_S)
        raw = self._inst.query('MREAD').strip()
        try:
            return int(raw)
        except ValueError:
            raise RuntimeError(f'MREAD invalid response @ {addr:#010x}: {raw!r}')
```

```python
# hp3458a_cal_dump.py — hozzáfűzni

def _fast_hypothesis_check(session: GpibSession) -> int | None:
    """3 MREAD-del ellenőrzi: A5 = 0x120000 ?
    
    BEEP 1→2 váltáskor A5+0x13A6 WORD értéke pontosan 256-tal nő
    (a high byte = BEEP mode, a low byte nem változik).
    """
    candidate = A5_HYPOTHESIS + BEEP_FLAG_OFFSET   # 0x1213A6

    session.beep(1)
    val_b1 = session.mread_word(candidate)

    session.beep(2)
    val_b2 = session.mread_word(candidate)

    session.beep(1)   # visszaállítás: MREAD aktív marad

    if val_b2 - val_b1 == 256:
        return A5_HYPOTHESIS
    return None
```

```python
# hp3458a_cal_dump.py — hozzáfűzni

def _coarse_fine_scan(session: GpibSession, start: int, end: int) -> int:
    """Két lépéses BEEP1/BEEP2 differenciál-scan az A5+0x13A6 cím megtalálásához."""
    COARSE = 0x100

    coarse_addrs = list(range(start & ~1, end & ~1, COARSE))

    session.beep(1)
    baseline = {a: session.mread_word(a) for a in coarse_addrs}

    session.beep(2)
    after    = {a: session.mread_word(a) for a in coarse_addrs}

    session.beep(1)

    hits = [a for a in coarse_addrs if after[a] - baseline[a] == 256]
    if not hits:
        raise RuntimeError(
            f'A5 nem azonosítható {start:#x}–{end:#x} tartományban. '
            'Próbáld --mode c opciót.')

    fine_start = max(start & ~1, hits[0] - COARSE)
    fine_end   = min(end   & ~1, hits[0] + COARSE)

    session.beep(1)
    fine_base  = {a: session.mread_word(a) for a in range(fine_start, fine_end, 2)}
    session.beep(2)
    fine_after = {a: session.mread_word(a) for a in range(fine_start, fine_end, 2)}
    session.beep(1)

    exact = [a for a in fine_base if fine_after[a] - fine_base[a] == 256]
    if not exact:
        raise RuntimeError('A5 fine-scan sikertelen.')

    return exact[0] - BEEP_FLAG_OFFSET


def detect_a5(session: GpibSession, mode: str) -> int:
    fast = _fast_hypothesis_check(session)
    if fast is not None:
        print(f'[A5] Hipotézis megerősítve: A5 = {fast:#010x}')
        return fast

    print('[A5] Hipotézis eltér, scan indul...')
    ranges = {'b': (0x11C000, 0x126000), 'c': (0x100000, 0x140000)}
    a5 = _coarse_fine_scan(session, *ranges[mode])
    print(f'[A5] Megtalálva: A5 = {a5:#010x}')
    return a5
```

```python
# hp3458a_cal_dump.py — hozzáfűzni

def compute_regions(a5: int) -> list[tuple[str, int, int]]:
    """Visszaadja a dump tartományokat: [(név, start_addr, word_count)]."""
    return [
        ('main_ram',  a5,         0x3000 // 2),   # 12 KB globális adat
        ('vectors',   0x000000,   0x0400 // 2),   # 1 KB 68000 vektorok
        ('maddr_reg', 0x0023F0,   1),              # MADDR regiszter (1 word)
    ]


def dump_regions(session: GpibSession,
                 regions: list[tuple[str, int, int]]) -> dict[int, int]:
    """Végigolvassa az összes régiót MREAD-del. Visszatér: {addr: signed_word}."""
    result: dict[int, int] = {}
    total = sum(cnt for _, _, cnt in regions)
    done  = 0

    for _, start, count in regions:
        for i in range(count):
            addr = start + i * 2
            result[addr] = session.mread_word(addr)
            done += 1
            if done % 200 == 0:
                pct = 100 * done // total
                print(f'\r  [{pct:3d}%] {done}/{total} word', end='', flush=True)

    print(f'\r  [100%] {total}/{total} word')
    return result
```

```python
# hp3458a_cal_dump.py — hozzáfűzni

def compute_hash(dump: dict[int, int]) -> str:
    """SHA-256 a rendezett (addr, word) párokból, big-endian kódolással."""
    h = hashlib.sha256()
    for addr in sorted(dump):
        h.update(struct.pack('>Ih', addr, dump[addr]))
    return h.hexdigest()


def verify_dumps(dumps: list[dict[int, int]]) -> tuple[bool, list[str]]:
    """Visszatér: (mind_egyezik, [hash1, hash2, hash3])."""
    hashes = [compute_hash(d) for d in dumps]
    return len(set(hashes)) == 1, hashes
```

```python
# hp3458a_cal_dump.py — hozzáfűzni

def save_json(dump: dict[int, int], metadata: dict, path: Path) -> None:
    data = {
        'metadata': metadata,
        'data': {f'0x{addr:08X}': val for addr, val in sorted(dump.items())},
    }
    path.write_text(json.dumps(data, indent=2))


def save_binary(dump: dict[int, int], path: Path) -> None:
    """Big-endian signed short-ok, cím szerint rendezve."""
    buf = b''.join(struct.pack('>h', dump[a]) for a in sorted(dump))
    path.write_bytes(buf)


def save_hexdump(dump: dict[int, int], path: Path) -> None:
    WORDS_PER_LINE = 8
    addrs = sorted(dump)
    lines = []
    for i in range(0, len(addrs), WORDS_PER_LINE):
        row = addrs[i:i + WORDS_PER_LINE]
        hex_parts, ascii_chars = [], []
        for a in row:
            w = dump[a] & 0xFFFF
            hex_parts.append(f'{w:04X}')
            for b in ((w >> 8) & 0xFF, w & 0xFF):
                ascii_chars.append(chr(b) if 0x20 <= b < 0x7F else '.')
        padding = '     ' * (WORDS_PER_LINE - len(row))
        lines.append(
            f'0x{row[0]:08X}: {" ".join(hex_parts)}{padding}  {"".join(ascii_chars)}'
        )
    path.write_text('\n'.join(lines) + '\n')


def save_outputs(dump: dict[int, int], metadata: dict,
                 outdir: Path, index: int) -> None:
    base = outdir / f'dump_{index:02d}'
    save_json(dump,    metadata, base.with_suffix('.json'))
    save_binary(dump,            base.with_suffix('.bin'))
    save_hexdump(dump,           base.with_suffix('.hex'))


def save_summary(a5: int, mode: str, hashes: list[str],
                 consistent: bool, outdir: Path) -> None:
    summary = {
        'timestamp':  datetime.now().isoformat(),
        'a5_base':    f'{a5:#010x}',
        'mode':       mode,
        'gpib':       GPIB_RESOURCE,
        'consistent': consistent,
        'hashes':     hashes,
    }
    (outdir / 'summary.json').write_text(json.dumps(summary, indent=2))
```

```python
# hp3458a_cal_dump.py — hozzáfűzni

def format_diff(baseline: dict[int, int], modified: dict[int, int]) -> str:
    header = ['# Differenciális elemzés',
              '# Cím           Előtte   Utána   Delta',
              '# ' + '─' * 40]
    body, changed = [], 0
    for addr in sorted(baseline):
        before = baseline[addr]
        after  = modified.get(addr, 0)
        if before != after:
            body.append(f'  0x{addr:08X}   {before:+7d}  {after:+7d}  {after-before:+7d}')
            changed += 1
    footer = [f'\n# Összesen: {changed} word változott']
    return '\n'.join(header + body + footer) + '\n'


def run_differential(session: GpibSession, baseline: dict[int, int],
                     outdir: Path, regions: list[tuple[str, int, int]]) -> None:
    input('\n[Diff] Változtass egy front-panel értéket (pl. NPLC), '
          'majd nyomj ENTER-t...')
    print('[Diff] Módosított dump folyamatban...')
    modified  = dump_regions(session, regions)
    diff_text = format_diff(baseline, modified)
    (outdir / 'diff.txt').write_text(diff_text)
    print(diff_text)
```

```python
def main() -> None:
    args   = parse_args()
    outdir = Path(f'dump_{datetime.now().strftime("%Y%m%d_%H%M%S")}')
    outdir.mkdir()
    print(f'[*] Kimenet könyvtár: {outdir}')

    with GpibSession() as session:
        print('[1/4] A5 detekció...')
        a5      = detect_a5(session, args.mode)
        regions = compute_regions(a5)
        total   = sum(c for _, _, c in regions)
        print(f'      A5 = {a5:#010x} | tartomány: {total} word')

        dumps = []
        for i in range(1, 4):
            print(f'\n[2/4] Dump {i}/3 ({total} word × 2 = {total*2} bájt)...')
            d = dump_regions(session, regions)
            meta = {
                'timestamp':   datetime.now().isoformat(),
                'a5_base':     f'{a5:#010x}',
                'mode':        args.mode,
                'gpib':        GPIB_RESOURCE,
                'dump_index':  i,
            }
            save_outputs(d, meta, outdir, i)
            dumps.append(d)

        print('\n[3/4] Hash-ellenőrzés...')
        consistent, hashes = verify_dumps(dumps)
```
